-- 1. users
CREATE TABLE users (
  id           BIGSERIAL PRIMARY KEY,
  username     VARCHAR(50) NOT NULL UNIQUE,
  email        VARCHAR(255) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  full_name    VARCHAR(255),
  created_at   TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at   TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- 2. accounts (wallets)
CREATE TABLE accounts (
  id           BIGSERIAL PRIMARY KEY,
  user_id      BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  name         VARCHAR(100) NOT NULL,
  currency     VARCHAR(10) DEFAULT 'INR',
  balance      NUMERIC(19,4) DEFAULT 0,
  created_at   TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at   TIMESTAMP WITH TIME ZONE DEFAULT now(),
  UNIQUE(user_id, name)
);

-- 3. categories
CREATE TABLE categories (
  id           BIGSERIAL PRIMARY KEY,
  user_id      BIGINT REFERENCES users(id) ON DELETE CASCADE,
  name         VARCHAR(100) NOT NULL,
  type         VARCHAR(10) NOT NULL CHECK (type IN ('INCOME','EXPENSE')),
  color        VARCHAR(7),
  created_at   TIMESTAMP WITH TIME ZONE DEFAULT now(),
  UNIQUE(user_id, name, type)
);

-- 4. transactions
CREATE TABLE transactions (
  id             BIGSERIAL PRIMARY KEY,
  user_id        BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  account_id     BIGINT NOT NULL REFERENCES accounts(id) ON DELETE CASCADE,
  category_id    BIGINT REFERENCES categories(id),
  amount         NUMERIC(19,4) NOT NULL CHECK (amount >= 0),
  type           VARCHAR(10) NOT NULL CHECK (type IN ('INCOME','EXPENSE','TRANSFER')),
  description    TEXT,
  transaction_at TIMESTAMP WITH TIME ZONE NOT NULL,  -- when the txn occurred
  created_at     TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at     TIMESTAMP WITH TIME ZONE DEFAULT now(),
  -- For transfer, store linked transaction id optional
  linked_txn_id  BIGINT
);

-- Indexes for common queries
CREATE INDEX idx_txn_user_date ON transactions (user_id, transaction_at);
CREATE INDEX idx_txn_account ON transactions (account_id);
CREATE INDEX idx_txn_category ON transactions (category_id);

-- 5. attachments (optional)
CREATE TABLE attachments (
  id           BIGSERIAL PRIMARY KEY,
  transaction_id BIGINT NOT NULL REFERENCES transactions(id) ON DELETE CASCADE,
  file_name    VARCHAR(255),
  file_size    BIGINT,
  s3_key       VARCHAR(512),
  created_at   TIMESTAMP WITH TIME ZONE DEFAULT now()
);
